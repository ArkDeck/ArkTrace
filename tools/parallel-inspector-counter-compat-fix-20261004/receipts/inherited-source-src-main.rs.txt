use arktrace_repository_inspector_parity::{Case, observe};
use std::{env, fs, path::Path};
fn main() {
    let arguments: Vec<String> = env::args().collect();
    assert_eq!(arguments.len(), 3);
    let cases: Vec<Case> = serde_json::from_slice(&fs::read(&arguments[2]).unwrap()).unwrap();
    assert!(!cases.is_empty() && cases.len() <= 32);
    let records = cases
        .iter()
        .map(|case| observe(case, Path::new(&arguments[1])))
        .collect::<Vec<_>>();
    println!("{}", serde_json::to_string_pretty(&records).unwrap());
}
